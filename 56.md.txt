#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;

    vector<vector<int>> g(n, vector<int>(n));
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> g[i][j];

    vector<int> path(n - 1);
    for (int i = 1; i < n; i++) path[i - 1] = i;

    int best = 1e9;

    do {
        int cost = 0, u = 0;
        for (int v : path) {
            cost += g[u][v];
            u = v;
        }
        cost += g[u][0];
        best = min(best, cost);
    } while (next_permutation(path.begin(), path.end()));

    cout << best;
}
