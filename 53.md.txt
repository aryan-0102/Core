#include <iostream>
#include <vector>
using namespace std;

vector<vector<int>> addM(vector<vector<int>>& A, vector<vector<int>>& B) {
    int n = A.size();
    vector<vector<int>> C(n, vector<int>(n));
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            C[i][j] = A[i][j] + B[i][j];
    return C;
}

vector<vector<int>> subM(vector<vector<int>>& A, vector<vector<int>>& B) {
    int n = A.size();
    vector<vector<int>> C(n, vector<int>(n));
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            C[i][j] = A[i][j] - B[i][j];
    return C;
}

vector<vector<int>> strassen(vector<vector<int>>& A, vector<vector<int>>& B) {
    int n = A.size();
    if (n == 1) {
        return {{A[0][0] * B[0][0]}};
    }

    int k = n / 2;

    vector<vector<int>> A11(k, vector<int>(k)), A12(k, vector<int>(k)),
                        A21(k, vector<int>(k)), A22(k, vector<int>(k)),
                        B11(k, vector<int>(k)), B12(k, vector<int>(k)),
                        B21(k, vector<int>(k)), B22(k, vector<int>(k));

    for (int i = 0; i < k; i++)
        for (int j = 0; j < k; j++) {
            A11[i][j] = A[i][j];
            A12[i][j] = A[i][j + k];
            A21[i][j] = A[i + k][j];
            A22[i][j] = A[i + k][j + k];
            B11[i][j] = B[i][j];
            B12[i][j] = B[i][j + k];
            B21[i][j] = B[i + k][j];
            B22[i][j] = B[i + k][j + k];
        }

    auto M1 = strassen(addM(A11, A22), addM(B11, B22));
    auto M2 = strassen(addM(A21, A22), B11);
    auto M3 = strassen(A11, subM(B12, B22));
    auto M4 = strassen(A22, subM(B21, B11));
    auto M5 = strassen(addM(A11, A12), B22);
    auto M6 = strassen(subM(A21, A11), addM(B11, B12));
    auto M7 = strassen(subM(A12, A22), addM(B21, B22));

    vector<vector<int>> C(n, vector<int>(n));

    vector<vector<int>> C11 = addM(subM(addM(M1, M4), M5), M7);
    vector<vector<int>> C12 = addM(M3, M5);
    vector<vector<int>> C21 = addM(M2, M4);
    vector<vector<int>> C22 = addM(subM(addM(M1, M3), M2), M6);

    for (int i = 0; i < k; i++)
        for (int j = 0; j < k; j++) {
            C[i][j] = C11[i][j];
            C[i][j + k] = C12[i][j];
            C[i + k][j] = C21[i][j];
            C[i + k][j + k] = C22[i][j];
        }

    return C;
}

int main() {
    int n;
    cin >> n;

    vector<vector<int>> A(n, vector<int>(n)), B(n, vector<int>(n));

    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> A[i][j];

    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> B[i][j];

    auto C = strassen(A, B);

    for (auto& row : C) {
        for (int x : row) cout << x << " ";
        cout << "\n";
    }
}
