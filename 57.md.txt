#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;

    vector<int> p(n);
    for (int i = 0; i < n; i++) cin >> p[i];

    vector<vector<int>> dp(n, vector<int>(n));

    for (int len = 2; len < n; len++) {
        for (int i = 0; i + len < n; i++) {
            int j = i + len;
            dp[i][j] = 1e9;
            for (int k = i + 1; k < j; k++) {
                int cost = dp[i][k] + dp[k][j] + p[i] * p[k] * p[j];
                if (cost < dp[i][j]) dp[i][j] = cost;
            }
        }
    }

    cout << dp[0][n - 1];
}
