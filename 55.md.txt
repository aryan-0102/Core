#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;

    vector<array<int,3>> jobs(n);
    for (int i = 0; i < n; i++)
        cin >> jobs[i][0] >> jobs[i][1] >> jobs[i][2];

    sort(jobs.begin(), jobs.end(), [](auto& a, auto& b){
        return a[2] > b[2];
    });

    int mx = 0;
    for (auto& j : jobs) mx = max(mx, j[1]);

    vector<int> slot(mx + 1, -1);
    int profit = 0;

    for (auto& j : jobs) {
        int id = j[0], d = j[1], p = j[2];
        for (int t = d; t >= 1; t--) {
            if (slot[t] == -1) {
                slot[t] = id;
                profit += p;
                break;
            }
        }
    }

    cout << profit;
}
