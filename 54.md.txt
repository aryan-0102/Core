#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int findP(vector<int>& p, int x) {
    if (p[x] == x) return x;
    return p[x] = findP(p, p[x]);
}

void unite(vector<int>& p, vector<int>& r, int a, int b) {
    a = findP(p, a);
    b = findP(p, b);
    if (a == b) return;
    if (r[a] < r[b]) swap(a, b);
    p[b] = a;
    if (r[a] == r[b]) r[a]++;
}

int main() {
    int n, m;
    cin >> n >> m;

    vector<array<int, 3>> edges(m);
    for (int i = 0; i < m; i++) cin >> edges[i][0] >> edges[i][1] >> edges[i][2];

    sort(edges.begin(), edges.end(), [](auto& a, auto& b){ return a[2] < b[2]; });

    vector<int> p(n), r(n);
    for (int i = 0; i < n; i++) p[i] = i;

    int cost = 0;
    for (auto& e : edges) {
        int u = e[0], v = e[1], w = e[2];
        if (findP(p, u) != findP(p, v)) {
            unite(p, r, u, v);
            cost += w;
        }
    }

    cout << cost;
}
