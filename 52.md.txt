#include <iostream>
#include <vector>
using namespace std;

int partitionIndex(vector<int>& a, int l, int r) {
    int p = a[r];
    int i = l - 1;
    for (int j = l; j < r; j++) {
        if (a[j] <= p) {
            i++;
            swap(a[i], a[j]);
        }
    }
    swap(a[i + 1], a[r]);
    return i + 1;
}

void quickSort(vector<int>& a, int l, int r) {
    if (l >= r) return;
    int pi = partitionIndex(a, l, r);
    quickSort(a, l, pi - 1);
    quickSort(a, pi + 1, r);
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    quickSort(a, 0, n - 1);

    for (int x : a) cout << x << " ";
}
