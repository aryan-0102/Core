#include <iostream>
#include <vector>
using namespace std;

int recursiveSearch(vector<int>& a, int l, int r, int x) {
    if (l > r) return -1;
    int mid = l + (r - l) / 2;
    if (a[mid] == x) return mid;
    if (x < a[mid]) return recursiveSearch(a, l, mid - 1, x);
    return recursiveSearch(a, mid + 1, r, x);
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    int x;
    cin >> x;

    int l = 0, r = n - 1;
    int ans = -1;

    while (l <= r) {
        int mid = l + (r - l) / 2;
        if (a[mid] == x) {
            ans = mid;
            break;
        } else if (x < a[mid]) r = mid - 1;
        else l = mid + 1;
    }

    cout << ans << '\n';
    cout << recursiveSearch(a, 0, n - 1, x) << '\n';
}
