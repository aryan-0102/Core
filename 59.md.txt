#include <iostream>
#include <vector>
using namespace std;

bool ok(int r, int c, vector<int>& col) {
    for (int i = 0; i < r; i++)
        if (col[i] == c || abs(col[i] - c) == abs(i - r)) return false;
    return true;
}

bool solve(int r, int n, vector<int>& col) {
    if (r == n) return true;
    for (int c = 0; c < n; c++) {
        if (ok(r, c, col)) {
            col[r] = c;
            if (solve(r + 1, n, col)) return true;
            col[r] = -1;
        }
    }
    return false;
}

int main() {
    int n;
    cin >> n;

    vector<int> col(n, -1);

    if (solve(0, n, col)) {
        for (int i = 0; i < n; i++)
            cout << col[i] << " ";
    } else {
        cout << -1;
    }
}
